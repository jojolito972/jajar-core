import os
import shutil

def backup_file(file_path, max_versions=4):
    if not os.path.exists(file_path):
        return # Rien à sauvegarder si le fichier n'existe pas encore
    
    # Rotation des sauvegardes (de max_versions-1 vers max_versions, etc.)
    for i in range(max_versions - 1, 0, -1):
        old_bak = f"{file_path}.bak.{i}"
        new_bak = f"{file_path}.bak.{i+1}"
        if os.path.exists(old_bak):
            if i + 1 == max_versions:
                os.remove(new_bak) if os.path.exists(new_bak) else None
            shutil.move(old_bak, new_bak)
            
    # Création du backup direct de la version actuelle (.bak.1)
    first_bak = f"{file_path}.bak.1"
    if os.path.exists(first_bak):
        # Si .bak.1 existe mais n'a pas été déplacé (cas où max_versions=1)
        pass
    shutil.copy2(file_path, first_bak)
    print(f"[ SÉCURITÉ ] Sauvegarde créée : {first_bak}")
```

"""
herdr_bridge.py
Pont entre un script Python d'agent JARVIS et Herdr (statut + envoi de tâches).
Mis en place par JARVIS (Superviseur Suprême).
"""
import os
import subprocess
import atexit

PANE_ID = os.environ.get("HERDR_PANE_ID")
IN_HERDR = os.environ.get("HERDR_ENV") == "1"

def report_state(agent_name: str, state: str, message: str | None = None):
    """
    Met à jour le statut visible dans la sidebar Herdr pour l'agent courant.
    state doit être : 'working', 'idle', 'blocked' ou 'unknown'.
    """
    if not IN_HERDR or not PANE_ID:
        return

    cmd = [
        "herdr", "pane", "report-agent", PANE_ID,
        "--source", f"custom:{agent_name}",
        "--agent", agent_name,
        "--state", state,
    ]
    if message:
        cmd += ["--message", message]

    try:
        subprocess.run(cmd, capture_output=True, timeout=5)
    except Exception:
        pass

def send_task(target_agent: str, text: str, wait: bool = False, timeout_ms: int = 60000):
    """
    Envoie une tâche à un autre agent dans son pane Herdr.
    """
    if wait:
        cmd = ["herdr", "agent", "prompt", target_agent, text, "--wait", "--timeout", str(timeout_ms)]
    else:
        cmd = ["herdr", "agent", "prompt", target_agent, text]

    result = subprocess.run(cmd, capture_output=True, text=True, timeout=(timeout_ms / 1000) + 5)
    return result.stdout

def read_agent_output(target_agent: str, lines: int = 80) -> str:
    """
    Lit ce qui est affiché dans le pane d'un autre agent.
    """
    cmd = ["herdr", "agent", "read", target_agent, "--source", "recent-unwrapped", "--lines", str(lines)]
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=10)
    return result.stdout

"""
telegram_state.py
Gestion de l'état persistant du mode vocal pour le bot Telegram de JARVIS.
"""
import json
import os

STATE_FILE = "data/telegram_state.json"

def _load_state() -> dict:
    if not os.path.exists(STATE_FILE):
        return {"vocal_mode": False}
    try:
        with open(STATE_FILE, "r") as f:
            return json.load(f)
    except Exception:
        return {"vocal_mode": False}

def _save_state(state: dict):
    os.makedirs(os.path.dirname(STATE_FILE), exist_ok=True)
    with open(STATE_FILE, "w") as f:
        json.dump(state, f)

def toggle_vocal_mode() -> bool:
    state = _load_state()
    state["vocal_mode"] = not state["vocal_mode"]
    _save_state(state)
    return state["vocal_mode"]

def is_vocal_mode() -> bool:
    return _load_state().get("vocal_mode", False)
```
